cmake_minimum_required(VERSION 3.24)
project(XSpeechListen VERSION 0.1.0 LANGUAGES C CXX)
set(XSL_RELEASE_VERSION "${PROJECT_VERSION}-beta.2")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
endif()

option(XSL_BUILD_PLUGINS "Build the native X-Plane plugins" ON)
option(XSL_LINUX_AVX2 "Tune the Linux development engine for a CPU with AVX2 and FMA" OFF)
option(XSL_VULKAN "Build the Vulkan inference back end" OFF)
set(XSL_TEST_MODELS "" CACHE PATH "Directory containing the pinned GGUF files")
include(CTest)
include(FetchContent)
find_package(Threads REQUIRED)

set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_INSTALL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_METAL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_VULKAN ${XSL_VULKAN} CACHE BOOL "" FORCE)
set(TRANSCRIBE_CUDA OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_USE_SYSTEM_BLAS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_X86_CONSERVATIVE ON CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
if(WIN32)
    set(TRANSCRIBE_BUILD_SHARED ON CACHE BOOL "" FORCE)
    set(TRANSCRIBE_GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
else()
    foreach(flag IN ITEMS SSE42 AVX AVX2 F16C FMA BMI2)
        set(GGML_${flag} ${XSL_LINUX_AVX2} CACHE BOOL "" FORCE)
    endforeach()
endif()

FetchContent_Declare(transcribe
    URL https://codeload.github.com/handy-computer/transcribe.cpp/tar.gz/cfa237c2551a4b2d3a56ff8fe5461e3c67c51772
    SOURCE_SUBDIR xsl-populate-only)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/55f93686c01528224f448c19128836e7df245f72)
FetchContent_Declare(miniaudio URL https://codeload.github.com/mackron/miniaudio/tar.gz/877ea4a7c2601c7b0f2a44e83dfac42e6470fea6 SOURCE_SUBDIR xsl-header-only)
FetchContent_MakeAvailable(transcribe json miniaudio)
# Configure the library after pruning, even when a build reuses populated sources.
# Keep a caller-supplied checkout intact.
if(FETCHCONTENT_SOURCE_DIR_TRANSCRIBE)
    set(transcribe_local "${CMAKE_CURRENT_BINARY_DIR}/transcribe-local-src")
    file(COPY "${transcribe_SOURCE_DIR}/" DESTINATION "${transcribe_local}" PATTERN ".git" EXCLUDE PATTERN "build" EXCLUDE)
    if(NOT EXISTS "${transcribe_SOURCE_DIR}/include/transcribe/listen.h")
        # These files are added by the patch and do not exist in the copied source.
        file(REMOVE "${transcribe_local}/include/transcribe/listen.h"
                    "${transcribe_local}/src/transcribe-listen-decode.h")
    endif()
    set(transcribe_SOURCE_DIR "${transcribe_local}")
endif()
include(cmake/ApplyTranscribeListen.cmake)
xsl_apply_transcribe_listen("${transcribe_SOURCE_DIR}")
include(cmake/PruneTranscribe.cmake)
xsl_prune_transcribe("${transcribe_SOURCE_DIR}")
add_subdirectory("${transcribe_SOURCE_DIR}" "${transcribe_BINARY_DIR}")
# The pinned upstream build uses CMAKE_SOURCE_DIR for this public include.
target_include_directories(transcribe PUBLIC "${transcribe_SOURCE_DIR}/include")
if(XSL_VULKAN)
    find_package(SPIRV-Headers CONFIG REQUIRED HINTS "$ENV{VULKAN_SDK}")
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()

add_library(xsl_core STATIC src/core.cpp src/process.cpp src/desktop.cpp)
target_include_directories(xsl_core PUBLIC include src)
target_link_libraries(xsl_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
if(WIN32)
    target_compile_definitions(xsl_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(xsl_core PUBLIC ole32 uuid user32)
endif()

add_executable(xsl-engine src/engine.cpp src/models.cpp src/audio.cpp src/fluid_weights.cpp)
include(cmake/FluidRecipes.cmake)
xsl_embed_fluid_recipes(xsl-engine)
target_include_directories(xsl-engine PRIVATE "${miniaudio_SOURCE_DIR}")
target_link_libraries(xsl-engine PRIVATE xsl_core transcribe ggml ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(xsl-engine PRIVATE winhttp bcrypt)
    set_target_properties(xsl-engine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    set_target_properties(transcribe PROPERTIES PREFIX "")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBCURL REQUIRED IMPORTED_TARGET libcurl)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(xsl-engine PRIVATE PkgConfig::LIBCURL OpenSSL::Crypto)
endif()

add_executable(xsl-check src/check.cpp)
target_link_libraries(xsl-check PRIVATE xsl_core)
if(MINGW)
    target_link_options(xsl-check PRIVATE -municode)
endif()

if(XSL_BUILD_PLUGINS)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/catalog.json")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/catalog.json" xsl_model_catalog)
    configure_file(cmake/model_catalog.hpp.in generated/model_catalog.hpp @ONLY)
    FetchContent_Declare(xpsdk URL https://codeload.github.com/X-Plane/xpsdk/tar.gz/ed88aed734106466436ba9efd9a87332fbd9808e)
    FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/tar.gz/f5befd2d29e66809cd1110a152e375a7f1981f06)
    FetchContent_MakeAvailable(xpsdk imgui)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS assets/locales/en.json)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/locales/en.json" xsl_english_catalog)
    configure_file(cmake/english_catalog.hpp.in generated/english_catalog.hpp @ONLY)
    add_library(xsl_imgui STATIC
        "${imgui_SOURCE_DIR}/imgui.cpp"
        "${imgui_SOURCE_DIR}/imgui_draw.cpp"
        "${imgui_SOURCE_DIR}/imgui_tables.cpp"
        "${imgui_SOURCE_DIR}/imgui_widgets.cpp"
        "${imgui_SOURCE_DIR}/misc/cpp/imgui_stdlib.cpp"
        src/window.cpp src/localization.cpp)
    target_link_libraries(xsl_imgui PRIVATE xsl_core nlohmann_json::nlohmann_json)
    target_include_directories(xsl_imgui PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_include_directories(xsl_imgui PUBLIC src
        "${imgui_SOURCE_DIR}" "${xpsdk_SOURCE_DIR}/CHeaders/XPLM")
    target_compile_definitions(xsl_imgui PUBLIC
        IMGUI_USE_WCHAR32 XPLM200=1 XPLM210=1 XPLM300=1 XPLM301=1 XPLM303=1 XPLM400=1 XPLM411=1 XPLM420=1 XPLM430=1 XPLM440=1)
    if(WIN32)
        target_compile_definitions(xsl_imgui PUBLIC IBM=1 NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(xsl_imgui PUBLIC "${xpsdk_SOURCE_DIR}/Libraries/Win/XPLM_64.lib")
        set(xsl_plugin_binary win)
    elseif(UNIX AND NOT APPLE)
        target_compile_definitions(xsl_imgui PUBLIC LIN=1)
        set(xsl_plugin_binary lin)
    else()
        message(FATAL_ERROR "This milestone targets Windows packages and Linux development")
    endif()

    foreach(plugin IN ITEMS provider diagnostic)
        add_library(xsl-${plugin} MODULE src/${plugin}.cpp)
        target_link_libraries(xsl-${plugin} PRIVATE xsl_core xsl_imgui)
        set_target_properties(xsl-${plugin} PROPERTIES
            PREFIX "" SUFFIX ".xpl" OUTPUT_NAME "${xsl_plugin_binary}"
            LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/plugins/${plugin}"
            RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/plugins/${plugin}")
    endforeach()
    target_sources(xsl-provider PRIVATE src/provider_microphone.cpp src/provider_storage.cpp src/provider_ui.cpp)
    target_compile_definitions(xsl-provider PRIVATE XSL_VERSION="${XSL_RELEASE_VERSION}")
    target_include_directories(xsl-provider PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(WIN32)
        install(TARGETS xsl-provider LIBRARY DESTINATION X-Speech-Listen/64 COMPONENT xsl
            RUNTIME DESTINATION X-Speech-Listen/64 COMPONENT xsl)
        install(TARGETS xsl-diagnostic LIBRARY DESTINATION X-Speech-Listen-Diagnostic/64 COMPONENT xsl
            RUNTIME DESTINATION X-Speech-Listen-Diagnostic/64 COMPONENT xsl)
        install(TARGETS xsl-engine transcribe ggml ggml-base RUNTIME DESTINATION X-Speech-Listen/bin COMPONENT xsl)
        get_property(ggml_targets DIRECTORY "${transcribe_SOURCE_DIR}/ggml/src" PROPERTY BUILDSYSTEM_TARGETS)
        foreach(backend IN LISTS ggml_targets)
            get_target_property(kind ${backend} TYPE)
            if(kind STREQUAL "MODULE_LIBRARY" AND backend MATCHES "^ggml-cpu-")
                install(FILES "$<TARGET_FILE:${backend}>" DESTINATION X-Speech-Listen/bin COMPONENT xsl)
            endif()
        endforeach()
        if(TARGET ggml-vulkan)
            install(FILES "$<TARGET_FILE:ggml-vulkan>" DESTINATION X-Speech-Listen/bin COMPONENT xsl)
        endif()
        install(TARGETS xsl-check RUNTIME DESTINATION tools COMPONENT xsl)
        install(DIRECTORY assets/locales DESTINATION X-Speech-Listen COMPONENT xsl)
        install(FILES assets/catalog.json assets/test-en.wav assets/test-fr.wav assets/test-es.wav assets/NotoSans-Regular.ttf
            DESTINATION X-Speech-Listen COMPONENT xsl)
        install(FILES include/xspeech_listen.h LICENSE README.md CHANGELOG.md
            DESTINATION . COMPONENT xsl)
        install(FILES "${transcribe_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME transcribe.cpp.txt COMPONENT xsl)
        install(FILES "${transcribe_SOURCE_DIR}/ggml/LICENSE" DESTINATION licenses RENAME ggml.txt COMPONENT xsl)
        install(FILES "${transcribe_SOURCE_DIR}/src/third_party/silero_vad/LICENSE"
            DESTINATION licenses RENAME silero-vad.txt COMPONENT xsl)
        install(FILES "${miniaudio_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME miniaudio.txt COMPONENT xsl)
        install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION licenses RENAME nlohmann-json.txt COMPONENT xsl)
        install(FILES "${imgui_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME dear-imgui.txt COMPONENT xsl)
        install(FILES assets/NotoSans-OFL.txt DESTINATION licenses COMPONENT xsl)
        install(FILES assets/MODEL-NOTICES.md DESTINATION licenses COMPONENT xsl)
    endif()
endif()

    add_executable(xsl-tests tests/core.cpp)
    target_link_libraries(xsl-tests PRIVATE xsl_core)
    add_test(NAME core COMMAND xsl-tests)
    add_test(NAME engine-process COMMAND xsl-check --engine $<TARGET_FILE:xsl-engine> --probe)
    set_tests_properties(engine-process PROPERTIES TIMEOUT 20)
    add_test(NAME microphone-test COMMAND xsl-check --engine $<TARGET_FILE:xsl-engine>
        --probe-microphone "${PROJECT_SOURCE_DIR}/assets/test-en.wav")
    set_tests_properties(microphone-test PROPERTIES TIMEOUT 20)
    if(XSL_BUILD_PLUGINS AND UNIX AND NOT APPLE)
        add_executable(xsl-native-host tests/native_host.cpp tests/locale_engine.cpp)
        set_target_properties(xsl-native-host PROPERTIES ENABLE_EXPORTS ON CXX_VISIBILITY_PRESET default)
        target_link_libraries(xsl-native-host PRIVATE xsl_core xsl_imgui ${CMAKE_DL_LIBS})
        add_test(NAME resize-grips COMMAND xsl-native-host --resize-grips)
        add_test(NAME native-sdk-host COMMAND xsl-native-host
            $<TARGET_FILE:xsl-provider> $<TARGET_FILE:xsl-diagnostic>
            $<TARGET_FILE:xsl-engine> "${PROJECT_SOURCE_DIR}/assets/catalog.json")
        add_test(NAME locale-ptt COMMAND xsl-native-host
            $<TARGET_FILE:xsl-provider> $<TARGET_FILE:xsl-diagnostic>
            --locale-ptt "${PROJECT_SOURCE_DIR}/assets/catalog.json")
        set_tests_properties(native-sdk-host locale-ptt PROPERTIES TIMEOUT 20)
    endif()
    if(XSL_TEST_MODELS)
        add_test(NAME real-models COMMAND xsl-check --engine $<TARGET_FILE:xsl-engine>
            --catalog "${PROJECT_SOURCE_DIR}/assets/catalog.json"
            --models "${XSL_TEST_MODELS}" --verify-files --wav "${PROJECT_SOURCE_DIR}/assets/test-en.wav")
        set_tests_properties(real-models PROPERTIES TIMEOUT 300 LABELS real-models)
    endif()
endif()
