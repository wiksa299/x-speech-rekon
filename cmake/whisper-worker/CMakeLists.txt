cmake_minimum_required(VERSION 3.24)
project(RekonWhisperWorker LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/utf-8)
endif()

set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_VULKAN ${XSR_VULKAN} CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(WIN32)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
else()
    foreach(flag IN ITEMS SSE42 AVX AVX2 F16C FMA BMI2)
        set(GGML_${flag} ${XSR_LINUX_AVX2} CACHE BOOL "" FORCE)
    endforeach()
endif()
add_subdirectory("${WHISPER_SOURCE}" upstream EXCLUDE_FROM_ALL)
if(XSR_VULKAN)
    find_package(SPIRV-Headers CONFIG REQUIRED HINTS "$ENV{VULKAN_SDK}")
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
    if(MINGW AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        # LLVM-MinGW spends disproportionate time optimizing this translation unit at -O3.
        set_property(SOURCE "${WHISPER_SOURCE}/ggml/src/ggml-vulkan/ggml-vulkan.cpp"
            TARGET_DIRECTORY ggml-vulkan APPEND PROPERTY COMPILE_OPTIONS "$<$<CONFIG:Release>:-O1>")
    endif()
endif()

find_package(Threads REQUIRED)
add_executable(xsr-whisper ../../src/whisper_worker.cpp ../../src/core.cpp ../../src/process.cpp
    "${WHISPER_SOURCE}/examples/grammar-parser.cpp")
target_include_directories(xsr-whisper PRIVATE ../../include ../../src
    "${JSON_SOURCE}/include" "${WHISPER_SOURCE}/examples")
target_link_libraries(xsr-whisper PRIVATE whisper ggml Threads::Threads)
if(WIN32)
    target_compile_definitions(xsr-whisper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
